#include <iostream>
using namespace std;

int main() {
    int length;
    cout << "Enter the size of the array: ";
    cin >> length;

    int* numbers = new int[length];

    cout << "Enter " << length << " elements:" << endl;
    for (int i = 0; i < length; i++) {
        cin >> numbers[i];
    }

    cout << "Fixed Heap Dynamic Array in C++:" << endl;
    for (int i = 0; i < length; i++) {
        cout << "Element " << i << ": " << numbers[i] << endl;
    }

    delete[] numbers;
    return 0;
}
