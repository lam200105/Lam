#include <iostream>

int main() {
    int length;
    std::cout << "Enter the size of the array: ";
    std::cin >> length;

    int* data = new int[length];

    std::cout << "Enter " << length << " elements:\n";
    for (int index = 0; index < length; ++index) {
        std::cin >> data[index];
    }

    std::cout << "Fixed Heap Dynamic Array in C++:\n";
    for (int index = 0; index < length; ++index) {
        std::cout << "Element " << index << ": " << data[index] << '\n';
    }

    delete[] data;
    return 0;
}
