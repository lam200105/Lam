#include <iostream>
using namespace std;

int main() {
    // Fixed stack array of size 5
    int arr[5] = {1, 2, 3, 4, 8};

    cout << "Fixed Stack Dynamic Array in C++:" << endl;
    for (int i = 0; i < 5; i++) {
        cout << "Element " << i << ": " << arr[i] << endl;
    }

    return 0;
}
